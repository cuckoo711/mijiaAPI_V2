# 最佳实践

## 凭据管理

- 用 `FileCredentialStore` / `create_auth_service` 存取，勿在代码里写明文 token
- 使用前检查 `is_expired()`，临期调用 `refresh_credential`（细节见[凭据刷新机制](06-凭据刷新机制.md)）
- 多用户时为每个用户独立凭据文件与客户端，可用 `create_multi_user_clients`

```python
from mijiaAPI_V2 import create_auth_service, create_api_client

auth_service = create_auth_service()
credential = auth_service.load_credential()
if credential and credential.is_expired():
    credential = auth_service.refresh_credential(credential)
    auth_service.save_credential(credential)
api = create_api_client(credential)
```

## 错误处理

按具体异常类型处理，读取失败可重试；控制抛出 `ExecutionUnknownError` 时，先核对实时状态或场景效果，禁止自动重发。异常类型与重试示例见[异常处理](../API参考/03-异常处理.md)。

### 读取失败时降级

```python
from mijiaAPI_V2.domain.exceptions import MijiaAPIException

def get_devices_with_fallback(api, home_id):
    """获取设备，失败时返回空列表"""
    try:
        return api.get_devices(home_id=home_id)
    except MijiaAPIException as e:
        logger.warning(f"获取设备失败，返回空列表: {e}")
        return []

def get_device_statistics_with_cache(api, home_id):
    """获取统计信息，失败时使用缓存"""
    cache_key = f"stats:{home_id}"
    
    try:
        stats = api.get_device_statistics(home_id)
        cache.set(cache_key, stats, ttl=3600)
        return stats
    except MijiaAPIException as e:
        logger.warning(f"获取统计失败，尝试使用缓存: {e}")
        cached = cache.get(cache_key)
        if cached:
            return cached
        # 返回默认值
        return {"total": 0, "online": 0, "offline": 0}
```

## 性能优化

### 1. 使用缓存

```python
# ✅ 好：利用缓存
def get_all_devices(api, home_ids):
    """获取多个家庭的设备（利用缓存）"""
    all_devices = []
    for home_id in home_ids:
        # 第一次从API获取，后续从缓存获取
        devices = api.get_devices(home_id)
        all_devices.extend(devices)
    return all_devices

# ❌ 不好：每次都清空缓存
def get_all_devices_no_cache(api, home_ids):
    all_devices = []
    for home_id in home_ids:
        cache.clear()  # 不要这样做！
        devices = api.get_devices(home_id)
        all_devices.extend(devices)
    return all_devices
```

### 2. 批量操作

```python
# ✅ 好：批量控制设备
def turn_on_all_lights(api, light_ids):
    """批量打开所有灯"""
    requests = [
        {"device_id": did, "siid": 2, "piid": 1, "value": True}
        for did in light_ids
    ]
    return api.batch_control_devices(requests)

# ❌ 不好：逐个控制
def turn_on_all_lights_slow(api, light_ids):
    results = []
    for did in light_ids:
        result = api.control_device(did, 2, 1, True)
        results.append(result)
    return results
```

### 3. 异步并发

```python
import asyncio

# ✅ 好：并发获取多个家庭的设备
async def get_all_devices_async(api, home_ids):
    """异步并发获取设备"""
    tasks = [api.get_devices(home_id) for home_id in home_ids]
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    # 过滤错误
    devices = []
    for result in results:
        if isinstance(result, Exception):
            logger.error(f"获取设备失败: {result}")
        else:
            devices.extend(result)
    
    return devices

# ❌ 不好：顺序执行
async def get_all_devices_slow(api, home_ids):
    devices = []
    for home_id in home_ids:
        result = await api.get_devices(home_id)
        devices.extend(result)
    return devices
```

```python
# ✅ 好：复用API客户端
class DeviceController:
```

## 日志记录

### 1. 结构化日志

```python
import logging
from mijiaAPI_V2.core.logging import get_logger

logger = get_logger(__name__)

def control_device_with_logging(api, device_id, siid, piid, value):
    """带日志的设备控制"""
    logger.info(
        "开始控制设备",
        extra={
            "device_id": device_id,
            "siid": siid,
            "piid": piid,
            "value": value
        }
    )
    
    try:
        result = api.control_device(device_id, siid, piid, value)
        
        logger.info(
            "设备控制成功",
            extra={
                "device_id": device_id,
                "result": result
            }
        )
        
        return result
    
    except Exception as e:
        logger.error(
            "设备控制失败",
            extra={
                "device_id": device_id,
                "error": str(e)
            },
            exc_info=True
        )
        raise
```

### 2. 性能监控

```python
import time
from contextlib import contextmanager

@contextmanager
def log_performance(operation: str):
    """记录操作性能"""
    start = time.time()
    try:
        yield
    finally:
        duration = time.time() - start
        logger.info(
            f"操作完成: {operation}",
            extra={"duration_ms": int(duration * 1000)}
        )

# 使用
with log_performance("获取设备列表"):
    devices = api.get_devices(home_id)
```

## 安全实践

### 1. 不要记录敏感信息

```python
# ✅ 好：脱敏日志
logger.info(f"用户登录: user_id={credential.user_id}")

# ❌ 不好：记录敏感信息
logger.info(f"用户登录: token={credential.service_token}")
```

### 2. 使用环境变量

```python
import os

# ✅ 好：从环境变量读取
redis_password = os.getenv("REDIS_PASSWORD")
api_key = os.getenv("API_KEY")

# ❌ 不好：硬编码
redis_password = "my_password"
api_key = "sk_live_1234567890"
```

### 3. 验证输入

```python
def control_device_safe(api, device_id, siid, piid, value):
    """安全的设备控制（验证输入）"""
    # 验证device_id
    if not device_id or not isinstance(device_id, str):
        raise ValueError("无效的设备ID")
    
    # 验证siid和piid
    if not isinstance(siid, int) or siid < 0:
        raise ValueError("无效的服务ID")
    
    if not isinstance(piid, int) or piid < 0:
        raise ValueError("无效的属性ID")
    
    # 执行控制
    return api.control_device(device_id, siid, piid, value)
```
